\section{Model}

We observe a count matrix $Y \in \mathbb{N}^{N \times M}$. Each row $y_i$ is modeled as
\[
  y_{im} \sim \Poi\Bigl(\sum_{d=1}^{D} \beta_{id}\, F_{\gamma_{id},\, m}\Bigr), \quad m \in [M],
\]
where $F \in \mathbb{R}_+^{K \times M}$ is a shared factor matrix satisfying $\sum_m F_{km} = 1$ for all $k$, and for each observation $i$ and effect $d$:
\begin{align}
  \gamma_{id} &\sim \Multi(1,\, \tfrac{1}{K}\mathbf{1}_K), \\
  \beta_{id} \mid \gamma_{id} &\sim \Gam(\alpha_{id}^0,\, \lambda_{id}^0).
\end{align}

\section{Variational family}

We use the mean-field variational family
\[
  q(\beta, \gamma) = \prod_{i=1}^{N} \prod_{d=1}^{D} q(\gamma_{id})\, q(\beta_{id} \mid \gamma_{id}),
\]
where $q(\gamma_{id}) = \Multi(1,\, \gb_{id})$ with $\gb_{id} = (\gb_{id1}, \ldots, \gb_{idK})$, and $q(\beta_{id} \mid \gamma_{id} = k) = \Gam(\alpha_{idk},\, \lambda_{idk})$.

We also introduce auxiliary variables $\xi_{imd} \geq 0$ with $\sum_d \xi_{imd} = 1$ for each $(i, m)$.

\section{Variational parameters}

The optimal variational parameters are:
\begin{align}
  \alpha_{idk} &= \sum_m \xi_{imd}\, y_{im} + \alpha_{id}^0, \label{eq:alpha} \\
  \lambda_{idk} &= \sum_m F_{km} + \lambda_{id}^0. \label{eq:lambda}
\end{align}
Note that $\alpha_{idk}$ does not depend on $k$, and $\lambda_{idk}$ does not depend on $i$.

\subsection{Moments}

Under $q(\beta_{id} \mid \gamma_{id} = k) = \Gam(\alpha_{idk}, \lambda_{idk})$:
\begin{align}
  \E_{q(\beta_{id} \mid \gamma_{id}=k)}[\log \beta_{id}] &= \psi(\alpha_{idk}) - \log(\lambda_{idk}), \label{eq:Elogbeta_cond} \\
  \E_{q(\beta_{id},\, \gamma_{id})}[\log \beta_{id}] &= \sum_k \gb_{idk}\bigl[\psi(\alpha_{idk}) - \log(\lambda_{idk})\bigr]. \label{eq:Elogbeta}
\end{align}

\section{ELBO}

The evidence lower bound is
\begin{align}
  \mathcal{E}(q) &= \sum_{i=1}^{N} \sum_{d=1}^{D} \E_{q_{id}(\gamma_{id},\, \beta_{id})} \Bigl[\sum_m \bigl(\xi_{imd}\, y_{im}(\log \beta_{id} + \log F_{\gamma_{id},\, m}) - \beta_{id}\, F_{\gamma_{id},\, m}\bigr)\Bigr] \nonumber \\
  &\quad - \sum_{i=1}^{N} \sum_{d=1}^{D} \KL\bigl(q(\gamma_{id},\, \beta_{id}) \,\|\, p(\gamma_{id},\, \beta_{id})\bigr) - \sum_{i,m,d} y_{im}\, \xi_{imd}\, \log \xi_{imd}. \label{eq:ELBO}
\end{align}

Expanding term by term:
\begin{align}
  \text{Term 1} &= \sum_{i,d,m} \xi_{imd}\, y_{im}\, \E_q[\log \beta_{id}], \\
  \text{Term 2} &= \sum_{i,d,m} \xi_{imd}\, y_{im} \sum_k \gb_{idk}\, \log F_{km}, \\
  \text{Term 3} &= -\sum_{i,d,m} \sum_k \gb_{idk}\, \frac{\alpha_{idk}}{\lambda_{idk}}\, F_{km}, \\
  \text{Term 4} &= -\sum_{i,m,d} y_{im}\, \xi_{imd}\, \log \xi_{imd}.
\end{align}

The KL divergence decomposes as $\KL = \KL_{\text{disc}} + \KL_{\text{cont}}$:
\begin{align}
  \KL_{\text{disc}} &= \sum_{i,d,k} \gb_{idk}\bigl(\log \gb_{idk} - \log \tfrac{1}{K}\bigr), \label{eq:KL_disc} \\
  \KL_{\text{cont}} &= \sum_{i,d,k} \gb_{idk} \Bigl[
    (\alpha_{idk} - \alpha_{id}^0)\,\psi(\alpha_{idk})
    + (\log \lambda_{idk} - \log \lambda_{id}^0)\,\alpha_{id}^0 \nonumber \\
    &\qquad\qquad\quad - \log\frac{\Gamma(\alpha_{idk})}{\Gamma(\alpha_{id}^0)}
    - (\lambda_{idk} - \lambda_{id}^0)\,\frac{\alpha_{idk}}{\lambda_{idk}}
  \Bigr]. \label{eq:KL_cont}
\end{align}

\section{Algorithm}

The current implementation uses several stabilization and speed options. When
$F$ is unknown, it can first run a standard Poisson/KL-NMF fit
$Y \approx LF$ with row-normalized $F$ and use the resulting $F$ as the initial
shared factor matrix. The corresponding NMF loading estimates can also
initialize $\gb_{id}$ by assigning the $D$ effects of observation $i$ to the
largest entries of the NMF loading row. During the MF-Poisson-SuSiE iterations,
the categorical update can be damped with step size $\rho_t$, and the $F$ update
can be damped with step size $\eta_t$. The default algorithm refreshes the
auxiliary responsibilities $\xi$ before each effect update, which is a
Gauss-Seidel-style schedule. The ELBO may be evaluated only every few
iterations because it is not needed for the coordinate updates.

\begin{algorithm}[H]
\caption{MF-Poisson-SuSiE with learned $F$}
\small
\begin{algorithmic}[1]
\State Initialize $F$ by either row-normalizing the supplied $F$ or by running
  Poisson/KL-NMF and row-normalizing the resulting factor rows.
\State Initialize $\gb$ either randomly or from the top entries of the
  Poisson-NMF loading rows.
\State Initialize $\alpha$, $\lambda$, and $\xi_{imd} = 1/D$.
\For{iter $= 1, \ldots, T$}
  \State Set damping weights $\rho_t$ for $\gb$ and $\eta_t$ for $F$.
  \State \textbf{(Break symmetry)} Merge duplicate effects if needed.
  \For{$d = 1, \ldots, D$}
    \State \textbf{E-step:} Refresh $\xi$ using all current effects:
    \[
      \xi_{imd} \leftarrow \text{Norm-Exp}_{d}\Bigl(
        \E_{q(\gamma_{id},\, \beta_{id})}
        [\log \beta_{id} + \log F_{\gamma_{id},\, m}]
      \Bigr), \quad \forall\, i \in [N],\, m \in [M].
    \]
    \State \textbf{M-step:} Update variational parameters for effect $d$:
    \begin{align*}
      \alpha_{idk} &\leftarrow \sum_m \xi_{imd}\, y_{im} + \alpha_{id}^0, \\
      \lambda_{idk} &\leftarrow \sum_m F_{km} + \lambda_{id}^0.
    \end{align*}
    \State \textbf{PIP:} Compute $\text{logprob}_{idk}$ and update $\gb_{id}$:
    \begin{align*}
      \text{logprob}_{idk} &= \sum_m \xi_{imd}\, y_{im}\, \log F_{km} - \alpha_{idk}\, \log \lambda_{idk}, \\
      \gb_{id}^{\mathrm{new}} &\leftarrow \text{Norm-Exp}(\text{logprob}_{id1}, \ldots, \text{logprob}_{idK}), \\
      \gb_{id} &\leftarrow (1-\rho_t)\gb_{id} + \rho_t \gb_{id}^{\mathrm{new}}.
    \end{align*}
    \State \textbf{VEB} (optional): Update
      $(\alpha_{id}^0,\lambda_{id}^0)$ using the empirical-Bayes equations in
      Section~\ref{sec:veb}.
  \EndFor
  \State Refresh $\xi$ using the final $\gb$, $\alpha$, and $\lambda$ from this
  outer iteration.
  \State \textbf{Update $F$} (optional): For each $k, m$:
  \begin{align*}
    A_{km} &= \sum_{i,d} \gb_{idk}\, \xi_{imd}\, y_{im}, \\
    F^{\mathrm{new}}_{km} &\leftarrow
      \frac{A_{km} + \epsilon}{\sum_{m'} (A_{km'} + \epsilon)}, \\
    F_{km} &\leftarrow (1-\eta_t)F_{km} + \eta_t F^{\mathrm{new}}_{km},
    \qquad
    F_{k\cdot} \leftarrow \frac{F_{k\cdot}}{\sum_{m'}F_{km'}}.
  \end{align*}
  \State Refresh $\xi$ after updating $F$.
  \If{ELBO evaluation is scheduled for this iteration}
    \State Evaluate $\mathcal{E}(q)$.
  \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Alternative $\xi$ refresh schedules]
Cheaper variants of the $\xi$ refresh are possible. A Jacobi-style schedule
refreshes $\xi$ once per outer iteration and reuses it for all $D$ effect
updates. A hybrid schedule uses the Gauss-Seidel refresh for an initial warm-up
period, then switches to the Jacobi-style refresh. These variants do not change
the model or the coordinate updates; they only trade off local optimization
behavior against speed.
\end{remark}

\section{Derivation of the $F$ update}

The terms in the ELBO involving $F_{km}$ are:
\[
  \sum_{k,m} \Bigl[ A_{km}\, \log F_{km} - B_k\, F_{km} \Bigr],
\]
where
\[
  A_{km} = \sum_{i,d} \gb_{idk}\, \xi_{imd}\, y_{im}, \qquad
  B_k = \sum_{i,d} \gb_{idk}\, \frac{\alpha_{idk}}{\lambda_{idk}}.
\]
Maximizing subject to $\sum_m F_{km} = 1$ via Lagrange multipliers:
\[
  \frac{\partial}{\partial F_{km}} \Bigl[ A_{km}\, \log F_{km} - B_k\, F_{km} - \mu_k\bigl(\sum_m F_{km} - 1\bigr) \Bigr] = 0
  \;\;\Longrightarrow\;\;
  F_{km} = \frac{A_{km}}{B_k + \mu_k}.
\]
Applying the constraint: $B_k + \mu_k = \sum_m A_{km}$, hence
\[
  F_{km} = \frac{A_{km}}{\sum_{m'} A_{km'}}.
\]

\section{Derivation of the VEB updates}\label{sec:veb}

The expected log-prior for effect $d$ of observation $i$ is:
\[
  \E_q[\log p(\beta_{id} \mid \gamma_{id};\, \alpha_{id}^0,\, \lambda_{id}^0)]
  = \sum_k \gb_{idk} \Bigl[ \alpha_{id}^0 \log \lambda_{id}^0 - \log \Gamma(\alpha_{id}^0) + (\alpha_{id}^0 - 1)\bigl(\psi(\alpha_{idk}) - \log \lambda_{idk}\bigr) - \lambda_{id}^0\, \frac{\alpha_{idk}}{\lambda_{idk}} \Bigr].
\]

\paragraph{$\lambda_{id}^0$ update (closed form).}
Setting $\frac{\partial}{\partial \lambda_{id}^0} = 0$:
\[
  \sum_k \gb_{idk} \Bigl[ \frac{\alpha_{id}^0}{\lambda_{id}^0} - \frac{\alpha_{idk}}{\lambda_{idk}} \Bigr] = 0
  \;\;\Longrightarrow\;\;
  \lambda_{id}^0 = \frac{\alpha_{id}^0}{\sum_k \gb_{idk}\, \alpha_{idk} / \lambda_{idk}}
  = \frac{\alpha_{id}^0}{\bar{\beta}_{id}}.
\]

\paragraph{$\alpha_{id}^0$ update (implicit equation).}
Setting $\frac{\partial}{\partial \alpha_{id}^0} = 0$:
\[
  \sum_k \gb_{idk} \Bigl[ \log \lambda_{id}^0 - \psi(\alpha_{id}^0) + \psi(\alpha_{idk}) - \log \lambda_{idk} \Bigr] = 0
  \;\;\Longrightarrow\;\;
  \psi(\alpha_{id}^0) = \overline{\log \beta}_{id} + \log \lambda_{id}^0.
\]
This is solved by inverting the digamma function, e.g., via Newton's method:
\[
  \alpha_{id}^0 \leftarrow \alpha_{id}^0 - \frac{\psi(\alpha_{id}^0) - \text{target}}{\psi'(\alpha_{id}^0)}, \qquad \text{target} = \overline{\log \beta}_{id} + \log \lambda_{id}^0.
\]
