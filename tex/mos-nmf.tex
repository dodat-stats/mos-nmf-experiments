\section{Purpose}

This note describes the target mixture-of-submanifolds NMF model in which the
submanifold motifs are random. It extends the current hard-motif implementation
by replacing the hard motif assignment
\[
  a_{sd} \in [K]
\]
with a random factor indicator
\[
  \gamma_{sd} \in [K],
\]
and by learning a posterior distribution over these indicators.

The goal is to quantify uncertainty in submanifold-level sparsity patterns.
This is especially important when several fitted factor rows are highly
correlated or when the number of fitted factors is overspecified.

\section{Model}

We observe a count matrix $Y \in \N^{N \times M}$. Let
$F \in \R_+^{K \times M}$ be a row-normalized factor matrix:
\[
  F_{km} \geq 0,
  \qquad
  \sum_{m=1}^{M} F_{km} = 1,
  \qquad k \in [K].
\]

There are $S$ submanifolds and each submanifold has nominal dimension $D$. For
submanifold $s$ and slot $d$, the factor identity is random:
\[
  \gamma_{sd} \sim \Cat(\rho_{sd1},\ldots,\rho_{sdK}).
\]
The simplest prior is uniform, $\rho_{sdk}=1/K$. More generally, $\rho_{sd}$ can
be learned or given a hierarchical prior.

For each observation $i$:
\begin{align}
  Z_i &\sim \Cat(\pi_1,\ldots,\pi_S), \\
  \beta_{id} \mid Z_i=s
    &\sim \Gam(\alpha^0_{sd},\lambda^0_{sd}),
    \qquad d \in [D], \\
  Y_{im} \mid Z_i=s,\gamma,\beta_i,F
    &\sim
    \Poi\left(
      \sum_{d=1}^{D} \beta_{id} F_{\gamma_{sd},m}
    \right),
    \qquad m \in [M].
\end{align}

Equivalently, the loading vector for observation $i$ is
\[
  L_i = \sum_{d=1}^{D} e_{\gamma_{Z_i d}}\beta_{id},
\]
where $e_k$ is the $k$th coordinate vector in $\R^K$.

\begin{remark}[Relationship to the hard-motif model]
The hard-motif model is recovered when each posterior or prior over
$\gamma_{sd}$ is a point mass. The hard motif $a_{sd}$ corresponds to
$q(\gamma_{sd}=a_{sd})=1$.
\end{remark}

\section{Variational family}

We use a mean-field approximation that keeps the submanifold motif indicators
global:
\[
  q(Z,\gamma,\beta)
  =
  \prod_{i=1}^{N} q(Z_i)
  \prod_{s=1}^{S}\prod_{d=1}^{D} q(\gamma_{sd})
  \prod_{i=1}^{N}\prod_{s=1}^{S}\prod_{d=1}^{D}
    q(\beta_{id}\mid Z_i=s).
\]
Write
\[
  q(Z_i=s)=\tz_{is},
  \qquad
  q(\gamma_{sd}=k)=\tg_{sdk},
  \qquad
  q(\beta_{id}\mid Z_i=s)=\Gam(\alpha_{isd},\lambda_{isd}).
\]

We introduce auxiliary variables
\[
  \xi_{isdm}\geq 0,
  \qquad
  \sum_{d=1}^{D}\xi_{isdm}=1,
\]
for each observation $i$, submanifold $s$, and coordinate $m$. These allocate
the count $Y_{im}$ across the $D$ slots of candidate submanifold $s$.

\section{Lower bound}

For fixed $i,s,m$, Jensen's inequality gives
\[
  \log\left(\sum_{d=1}^{D}\beta_{id}F_{\gamma_{sd},m}\right)
  \geq
  \sum_{d=1}^{D}
  \xi_{isdm}
  \log\left(
    \frac{\beta_{id}F_{\gamma_{sd},m}}{\xi_{isdm}}
  \right).
\]
Ignoring constants $-\log(Y_{im}!)$, the variational lower bound is
\begin{align}
  \mathcal{L}
  &=
  \sum_{i,s}\tz_{is}
  \sum_{d,m}
  \xi_{isdm}Y_{im}
  \left\{
    \E_q[\log \beta_{id}\mid Z_i=s]
    +
    \E_q[\log F_{\gamma_{sd},m}]
    -
    \log \xi_{isdm}
  \right\} \nonumber \\
  &\quad
  -
  \sum_{i,s}\tz_{is}
  \sum_{d,m}
  \E_q[\beta_{id}\mid Z_i=s]\,
  \E_q[F_{\gamma_{sd},m}]
  \nonumber \\
  &\quad
  -
  \sum_{i,s}\tz_{is}\sum_d
  \KL\{q(\beta_{id}\mid Z_i=s)\,\|\,p(\beta_{id}\mid Z_i=s)\}
  \nonumber \\
  &\quad
  -
  \sum_{s,d}
  \KL\{q(\gamma_{sd})\,\|\,p(\gamma_{sd})\}
  +
  \sum_{i,s}\tz_{is}(\log \pi_s-\log \tz_{is}).
  \label{eq:soft-mos-elbo}
\end{align}
The expectations over $\gamma$ are
\begin{align}
  \E_q[\log F_{\gamma_{sd},m}]
    &= \sum_{k=1}^{K}\tg_{sdk}\log F_{km}, \\
  \E_q[F_{\gamma_{sd},m}]
    &= \sum_{k=1}^{K}\tg_{sdk}F_{km}.
\end{align}

\section{Coordinate updates}

\subsection{Auxiliary allocation}

For each $i,s,m,d$,
\[
  \xi_{isdm}
  \leftarrow
  \frac{
    \exp\left\{
      \E_q[\log\beta_{id}\mid Z_i=s]
      +
      \sum_k \tg_{sdk}\log F_{km}
    \right\}
  }{
    \sum_{d'=1}^{D}
    \exp\left\{
      \E_q[\log\beta_{id'}\mid Z_i=s]
      +
      \sum_k \tg_{sd'k}\log F_{km}
    \right\}
  }.
\]

\subsection{Gamma effect-size posterior}

For each $i,s,d$,
\begin{align}
  \alpha_{isd}
    &\leftarrow
    \alpha^0_{sd}
    +
    \sum_{m=1}^{M}\xi_{isdm}Y_{im}, \\
  \lambda_{isd}
    &\leftarrow
    \lambda^0_{sd}
    +
    \sum_{m=1}^{M}\sum_{k=1}^{K}\tg_{sdk}F_{km}.
\end{align}
With row-normalized $F$, the second term is $1$ for any valid probability
vector $\tg_{sd}$, but keeping the expression explicit is useful for checking
scale conventions.

\subsection{Submanifold motif posterior}

For each submanifold $s$, slot $d$, and candidate factor $k$, collect the terms
in the ELBO that depend on $\gamma_{sd}=k$. Dropping constants in $k$ gives
\begin{align}
  \ell_{sdk}
  &=
  \log \rho_{sdk}
  +
  \sum_{i=1}^{N}\tz_{is}
  \left[
    \sum_{m=1}^{M}
      \xi_{isdm}Y_{im}\log F_{km}
    -
    \frac{\alpha_{isd}}{\lambda_{isd}}
    \sum_{m=1}^{M}F_{km}
  \right].
\end{align}
Then
\[
  \tg_{sdk}
  \leftarrow
  \frac{\exp(\ell_{sdk})}
       {\sum_{\ell=1}^{K}\exp(\ell_{sd\ell})}.
\]
If $F$ is row-normalized, the term
$\sum_m F_{km}$ is constant in $k$ and drops out. In that case the update is
driven by responsibility-weighted allocated counts matched against
$\log F_{k\cdot}$, plus the prior $\rho_{sd}$.

\begin{remark}[Posterior uncertainty in motifs]
The vector $\tg_{sd}=(\tg_{sd1},\ldots,\tg_{sdK})$ is the submanifold-level
analog of a SuSiE PIP vector. Diffuse $\tg_{sd}$ means that slot $d$ of
submanifold $s$ is uncertain across several factor rows. This is the object that
should quantify uncertainty when overfitted or highly correlated factor rows
compete to explain the same sparsity pattern.
\end{remark}

\subsection{Mixture responsibilities}

Define the component lower bound contribution for observation $i$ under
submanifold $s$:
\begin{align}
  \mathcal{L}_{is}
  &=
  \sum_{d,m}
  \xi_{isdm}Y_{im}
  \left\{
    \psi(\alpha_{isd})-\log\lambda_{isd}
    +
    \sum_k\tg_{sdk}\log F_{km}
    -
    \log \xi_{isdm}
  \right\} \nonumber \\
  &\quad
  -
  \sum_{d,m}
  \frac{\alpha_{isd}}{\lambda_{isd}}
  \sum_k\tg_{sdk}F_{km}
  -
  \sum_d
  \KL\{\Gam(\alpha_{isd},\lambda_{isd})
        \,\|\,\Gam(\alpha^0_{sd},\lambda^0_{sd})\}.
\end{align}
Then
\[
  \tz_{is}
  \leftarrow
  \frac{\pi_s\exp(\mathcal{L}_{is})}
       {\sum_{s'=1}^{S}\pi_{s'}\exp(\mathcal{L}_{is'})}.
\]
The mixture weights are updated by
\[
  \pi_s \leftarrow \frac{1}{N}\sum_{i=1}^{N}\tz_{is}.
\]

\subsection{Empirical-Bayes prior update}

For each submanifold $s$ and slot $d$, compute responsibility-weighted moments
\begin{align}
  \bar{\beta}_{sd}
  &=
  \frac{\sum_i \tz_{is}\alpha_{isd}/\lambda_{isd}}
       {\sum_i \tz_{is}}, \\
  \overline{\log\beta}_{sd}
  &=
  \frac{\sum_i \tz_{is}\{\psi(\alpha_{isd})-\log\lambda_{isd}\}}
       {\sum_i \tz_{is}}.
\end{align}
The Gamma shape solves
\[
  \log\alpha^0_{sd}-\psi(\alpha^0_{sd})
  =
  \log\bar{\beta}_{sd}-\overline{\log\beta}_{sd},
\]
and
\[
  \lambda^0_{sd}\leftarrow \alpha^0_{sd}/\bar{\beta}_{sd}.
\]

\subsection{Factor update}

The terms involving $F_{km}$ are weighted by the posterior probability that a
submanifold slot uses factor $k$. The expected allocated count is
\[
  A_{km}
  =
  \sum_{i=1}^{N}\sum_{s=1}^{S}\sum_{d=1}^{D}
  \tz_{is}\tg_{sdk}\xi_{isdm}Y_{im}.
\]
With row-normalized $F$, the update is
\[
  F_{km}^{\mathrm{new}}
  \leftarrow
  \frac{A_{km}+\epsilon}
       {\sum_{m'=1}^{M}(A_{km'}+\epsilon)}.
\]
A damped update can be used:
\[
  F_{k\cdot}
  \leftarrow
  (1-\eta_t)F_{k\cdot}+\eta_t F^{\mathrm{new}}_{k\cdot},
  \qquad
  F_{k\cdot}\leftarrow F_{k\cdot}/\sum_mF_{km}.
\]

\section{Algorithm}

\begin{algorithm}[H]
\caption{Soft-motif MoS-NMF}
\small
\begin{algorithmic}[1]
\State Initialize $F$, for example by MF-Poisson-SuSiE with learned $F$.
\State Initialize $\tz_{is}$ by clustering posterior loading scores or by
random assignment.
\State Initialize $\tg_{sdk}$ from cluster-level loading patterns, from hard
motifs, or randomly.
\State Initialize $\alpha_{isd}$, $\lambda_{isd}$, and $\xi_{isdm}=1/D$.
\For{$t=1,\ldots,T$}
  \For{$s=1,\ldots,S$}
    \For{$d=1,\ldots,D$}
      \State Update $\xi_{isdm}$ for all $i,m$ using current $\tg$, $\alpha$,
      and $\lambda$.
      \State Update $\alpha_{isd}$ and $\lambda_{isd}$ for all $i$.
      \State Update motif posterior $\tg_{sdk}$ for all $k$.
      \State Optionally update $(\alpha^0_{sd},\lambda^0_{sd})$.
    \EndFor
  \EndFor
  \State Update component lower bounds $\mathcal{L}_{is}$.
  \State Update mixture responsibilities $\tz_{is}$ and mixture weights $\pi_s$.
  \State Optionally update $F$ using expected counts
  $A_{km}=\sum_{i,s,d}\tz_{is}\tg_{sdk}\xi_{isdm}Y_{im}$.
  \State Evaluate or monitor the ELBO in Eq.~\eqref{eq:soft-mos-elbo}.
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Relationship to current implementation}

The current implemented hard-motif algorithm uses a matrix
$a_{sd}\in[K]$ and responsibilities $\omega_{is}$. It corresponds to the
restricted variational family
\[
  \tg_{sdk}=\mathbf{1}\{k=a_{sd}\}.
\]
The soft-motif model replaces this point mass with a full categorical posterior
for each submanifold slot. Under this replacement:
\begin{itemize}
  \item the hard count update for $F$ becomes an expected count update weighted
  by $\tg_{sdk}$;
  \item hard motif changes become categorical posterior updates;
  \item uncertainty in overfitted or correlated factor rows is represented by
  diffuse $\tg_{sd}$.
\end{itemize}

\section{Open implementation choices}

\begin{remark}[Observation-specific versus submanifold-specific effects]
The model above keeps $\gamma_{sd}$ global to submanifold $s$ while
$\beta_{id}$ remains observation-specific. This is the intended
submanifold-level sparsity pattern model. An alternative would give each
observation its own $\gamma_{isd}$ while tying it through a submanifold-level
prior, but that would be closer to a hierarchical MF-Poisson-SuSiE model than to
a motif model.
\end{remark}

\begin{remark}[Avoiding duplicated motif slots]
Independent categorical posteriors over $\gamma_{sd}$ allow two slots in the
same submanifold to choose the same factor. This is useful when the effective
dimension is smaller than $D$, but it can also duplicate active slots. Possible
regularizations include symmetry breaking, repulsive priors, or a
without-replacement motif prior.
\end{remark}

\begin{remark}[Monotonicity]
If all updates are derived from the same lower bound and are performed exactly,
the coordinate ascent should be monotone. In practice, damping, initialization
heuristics, symmetry breaking, or approximate updates may make the monitored
objective non-monotone.
\end{remark}
